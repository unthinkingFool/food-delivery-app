-- ============================================================
-- Food Delivery App — PostgreSQL Schema
-- ============================================================

-- Drop tables if they exist (in dependency-safe order)
DROP TABLE IF EXISTS PAYMENT CASCADE;
DROP TABLE IF EXISTS REVIEW CASCADE;
DROP TABLE IF EXISTS ORDER_ITEM CASCADE;
DROP TABLE IF EXISTS FOOD_ORDER CASCADE;
DROP TABLE IF EXISTS LOCATION CASCADE;
DROP TABLE IF EXISTS ITEM CASCADE;
DROP TABLE IF EXISTS CUSTOMER CASCADE;
DROP TABLE IF EXISTS RIDER CASCADE;
DROP TABLE IF EXISTS RESTAURANT CASCADE;
DROP TABLE IF EXISTS ADMIN CASCADE;

DROP TYPE IF EXISTS restaurant_status_enum;
DROP TYPE IF EXISTS order_status_enum;
DROP TYPE IF EXISTS item_type_enum;
DROP TYPE IF EXISTS payment_provider_enum;
DROP TYPE IF EXISTS payment_method_enum;

-- ============================================================
-- ENUM TYPES
-- (values are reasonable defaults for a food-delivery domain —
--  adjust the value lists to match your actual business rules)
-- ============================================================

-- RESTAURANT.status — operational state set by the restaurant/admin
CREATE TYPE restaurant_status_enum AS ENUM ('open', 'closed', 'notApproved');

-- FOOD_ORDER.status — order lifecycle
CREATE TYPE order_status_enum AS ENUM (
    'pending',
    'confirmed',
    'preparing',
    'out_for_delivery',
    'delivered',
    'cancelled'
);

-- ITEM.type — dietary/menu category
CREATE TYPE item_type_enum AS ENUM ('burger', 'pizza', 'drink','fries');

-- PAYMENT.payment_provider — who processed the payment
CREATE TYPE payment_provider_enum AS ENUM (
    'bkash',
    'nagad',
    'rocket',
    'card',
    'cash_on_delivery'
);

-- PAYMENT.method — how the payment was made
CREATE TYPE payment_method_enum AS ENUM ('cash', 'card', 'mobile_banking');

-- ============================================================
-- ADMIN
-- ============================================================
CREATE TABLE ADMIN (
    email           VARCHAR(255) PRIMARY KEY,
    hashed_password VARCHAR(255) NOT NULL
);

-- ============================================================
-- RESTAURANT
-- ============================================================
CREATE TABLE RESTAURANT (
    id              SERIAL PRIMARY KEY,
    isApproved      BOOLEAN NOT NULL DEFAULT FALSE,
    status          restaurant_status_enum NOT NULL DEFAULT 'notApproved',
    name            VARCHAR(255) NOT NULL,
    image_link       VARCHAR(500),
    description     TEXT,
    address         VARCHAR(255),
    city            VARCHAR(100) NOT NULL,
    email           VARCHAR(255) NOT NULL UNIQUE,
    hashed_password VARCHAR(255) NOT NULL,
    contact_no      VARCHAR(50),
    rating          NUMERIC(3,2),
    created_at      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CHECK (rating IS NULL OR rating BETWEEN 0 AND 5)
);

-- ============================================================
-- RIDER
-- ============================================================
CREATE TABLE RIDER (
    id              SERIAL PRIMARY KEY,
    isApproved      BOOLEAN NOT NULL DEFAULT FALSE,
    name            VARCHAR(255) NOT NULL,
    email           VARCHAR(255) NOT NULL UNIQUE,
    hashed_password VARCHAR(255) NOT NULL,
    contact_no      VARCHAR(50),
    income          NUMERIC(12,2),
    rating          NUMERIC(3,2),
    latitude        DECIMAL(9,6),
    longitude       DECIMAL(9,6),
    CHECK (rating IS NULL OR rating BETWEEN 0 AND 5)
);

-- ============================================================
-- CUSTOMER
-- ============================================================
CREATE TABLE CUSTOMER (
    id              SERIAL PRIMARY KEY,
    name            VARCHAR(255) NOT NULL,
    email           VARCHAR(255) NOT NULL UNIQUE,
    hashed_password VARCHAR(255) NOT NULL,
    phone_number    VARCHAR(50) NOT NULL,
    created_at      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- ITEM  (RESTAURANT "offers" ITEM)
-- ============================================================
CREATE TABLE ITEM (
    id              SERIAL PRIMARY KEY,
    restaurant_id   INTEGER NOT NULL REFERENCES RESTAURANT(id) ON DELETE CASCADE,
    name            VARCHAR(255) NOT NULL,
    type            item_type_enum,
    description     VARCHAR(500),
    price           NUMERIC(10,2) NOT NULL,
    discount_price  NUMERIC(10,2),
    image_link       VARCHAR(500),
    total_sold      INTEGER DEFAULT 0,
    rating          NUMERIC(3,2),
    created_at      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    isAvailable     BOOLEAN NOT NULL DEFAULT TRUE,
    CHECK (price >= 0),
    CHECK (discount_price IS NULL OR discount_price >= 0),
    CHECK (rating IS NULL OR rating BETWEEN 0 AND 5)
);

-- ============================================================
-- LOCATION  (CUSTOMER "resides at" LOCATION)
-- ============================================================
CREATE TABLE LOCATION (
    id              SERIAL PRIMARY KEY,
    customer_id     INTEGER NOT NULL REFERENCES CUSTOMER(id),
    road            VARCHAR(255),
    city            VARCHAR(100)
);

-- ============================================================
-- FOOD_ORDER
-- (CUSTOMER "places", RESTAURANT "receives", RIDER "delivers")
-- ============================================================
CREATE TABLE FOOD_ORDER (
    id              SERIAL PRIMARY KEY,
    customer_id     INTEGER NOT NULL REFERENCES CUSTOMER(id),
    restaurant_id   INTEGER NOT NULL REFERENCES RESTAURANT(id),
    rider_id        INTEGER REFERENCES RIDER(id),
    total_price     INTEGER NOT NULL,
    requirement     VARCHAR(500),
    status          order_status_enum NOT NULL DEFAULT 'pending',
    created_at      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CHECK (total_price >= 0)
);

-- ============================================================
-- ORDER_ITEM
-- (FOOD_ORDER "contains", ITEM "included in")
-- ============================================================
CREATE TABLE ORDER_ITEM (
    id              SERIAL PRIMARY KEY,
    order_id        INTEGER NOT NULL REFERENCES FOOD_ORDER(id),
    item_id         INTEGER NOT NULL REFERENCES ITEM(id),
    quantity        INTEGER NOT NULL,
    CHECK (quantity > 0)
);

-- ============================================================
-- REVIEW
-- (ITEM "receives", CUSTOMER "writes")
-- ============================================================
CREATE TABLE REVIEW (
    id              SERIAL PRIMARY KEY,
    item_id         INTEGER NOT NULL REFERENCES ITEM(id),
    customer_id     INTEGER NOT NULL REFERENCES CUSTOMER(id),
    rating          SMALLINT NOT NULL,
    description     VARCHAR(500),
    CHECK (rating BETWEEN 1 AND 5)
);

-- ============================================================
-- PAYMENT  (FOOD_ORDER "pay" PAYMENT)
-- ============================================================
CREATE TABLE PAYMENT (
    id                SERIAL PRIMARY KEY,
    order_id          INTEGER NOT NULL REFERENCES FOOD_ORDER(id),
    payment_provider  payment_provider_enum,
    method            payment_method_enum,
    transaction_id    VARCHAR(255),
    paid_at           TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

--=============================================================
-- Forget Password
--=============================================================
CREATE TABLE PASSWORD_RESET (
    id SERIAL PRIMARY KEY,
    email VARCHAR(255) NOT NULL,
    role VARCHAR(20) NOT NULL,
    otp_hash VARCHAR(255) NOT NULL,
    expires_at TIMESTAMP NOT NULL,
    verified BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
    CHECK (role IN ('customer', 'restaurant', 'rider'))
);

--=============================================================
-- Notification
-- ============================================================
CREATE TABLE NOTIFICATION (
    id SERIAL PRIMARY KEY,
    recipient_role VARCHAR(20) NOT NULL,
    recipient_id INTEGER NOT NULL,
    type VARCHAR(50) NOT NULL,
    title VARCHAR(255) NOT NULL,
    message VARCHAR(500) NOT NULL,
    reference_id INTEGER,
    is_read BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);
